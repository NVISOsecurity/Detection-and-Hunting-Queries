# Suspicious Process Spawned by Web Service Process - PHP2Shell Post-Exploitation - CVE-2026-109618, CVE-2026-109619

## Description

This query identifies known suspicious processes user by adversaries or known suspicious process patterns spawned by Web Service processes. Based on recent PHP2Shell CVE-2026-109618, CVE-2026-109619, were an exploitation of PHP versions can lead to remote code execution (RCE), this query can be used for threat hunting purposes to identify suspicious processes spawned by Web Service processes. The following queries are extended to multiple web service processes and not only necessarily php related processes.

Query for Linux based threat hunting:

```
let common_webserver_executables_linux = dynamic([@'php-fpm', @'php-cgi', @'php', @'lsphp', @'frankenphp', @'artisan', @'phpdbg', @'composer', @'phpunit', @'rr', @'roadrunner', @'httpd', @'apache2', @'apachectl', @'nginx', @'caddy', @'lshttpd', @'lighttpd', @'hiawatha', @'cherokee-worker', @'cherokeed', @'varnishd', @'haproxy', @'traefik', @'envoy', @'squid', @'gunicorn', @'uwsgi', @'uvicorn', @'openlitespeed']);
let SuspiciousCommands = dynamic(["busybox", "python", "perl", "ruby", "base64_decode", "gzinflate", "str_rot13", "base64"]);
let ShellCommands = dynamic(["sh", "bash", "dash", "zsh", "ksh"]);
let ReconCommands = dynamic(["id ", "whoami", "who", "last", "lastlog", "hostname", "uname -a", "iptables", "ifconfig", "cat /etc/os-release", "uptime", "getent", "ldapsearch", "/etc/os-release", "env", "printenv", "history", ".bash_history", ".zsh_history", "ip addr", "ip route", "ifconfig"]);
let DownloaderCommands = dynamic(["curl", "wget", "fetch", "aria2c", "nc ", "ncat", "netcat", "socat"]);
let PersistenceCommands = dynamic(["crontab", "systemctl enable", "systemctl start", "systemctl daemon-reload", "service", "mkdir -p ~/.ssh", ".ssh/authorized_keys", "chmod 700 ~/.ssh", "chmod 600 ~/.ssh/authorized_keys", ".bashrc", ".bash_profile", ".profile", ".zshrc"]);
let ReverseShellPatternsCommands = dynamic(["fsockopen", "nc ", "ncat", "netcat", "socat", "telnet", "openssl s_client", "/dev/tcp/", "/dev/udp/", "nc -e", "ncat -e", "mkfifo", "socket(", "connect(", "dup2(", "pty.spawn("]);
let WebShellsCommands = dynamic(["php -r", "php -f", "php /tmp/", "php /var/tmp/", "php /dev/shm/", "php -d auto_prepend_file=", "php -d allow_url_include=1", "eval(", "assert(", "shell_exec", "system(", "exec(", "passthru", "proc_open", "popen", "php artisan queue:work", "php artisan queue:listen", "php artisan horizon", "php artisan schedule:work", "php artisan octane:start", "php bin/console messenger:consume"]);
let DefenseEvasionCommands = dynamic(["/dev/null", "history -c", "unset HISTFILE", "HISTFILE=/dev/null", "/.bash_history", "shred -u", "truncate -s 0 /var/log/", "journalctl --vacuum-time=", "systemctl stop", "systemctl disable", "ufw disable", "iptables", "nft flush ruleset", "setenforce 0"]);
let ExfiltrationMiningCommands = dynamic(["tar", "7z", "zip", "unzip", "wget --post-file", "scp", "rsync", "sftp", "ftp", "xmrig", "minerd", "cpuminer", "kinsing", "kthreaddk", "watchdogd"]);
//Exclude known or expected processes on each environment
let NoisyCommandsExclussion_ProcessCommandLine = dynamic([]);
let NoisyCommandsExclussion_InitiatingProcessCommandLine = dynamic([]);
DeviceProcessEvents
| where Timestamp >= ago(30d)
| where FolderPath startswith @"/"
| where InitiatingProcessFileName has_any (common_webserver_executables_linux) or InitiatingProcessParentFileName has_any (common_webserver_executables_linux)
| where ProcessCommandLine has_any (SuspiciousCommands)
    or ProcessCommandLine has_any (ShellCommands)
    or ProcessCommandLine has_any (ReconCommands)
    or ProcessCommandLine has_any (DownloaderCommands)
    or ProcessCommandLine has_any (PersistenceCommands)
    or ProcessCommandLine has_any (ReverseShellPatternsCommands)
    or ProcessCommandLine has_any (WebShellsCommands)
    or ProcessCommandLine has_any (DefenseEvasionCommands)
    or ProcessCommandLine has_any (ExfiltrationMiningCommands)
| project-reorder InitiatingProcessFileName, InitiatingProcessCommandLine, ProcessCommandLine
| where not(ProcessCommandLine has_any (NoisyCommandsExclussion_ProcessCommandLine)) and not(InitiatingProcessCommandLine has_any (NoisyCommandsExclussion_InitiatingProcessCommandLine))
| summarize
    count(),
    min(Timestamp),
    max(Timestamp),
    make_set(DeviceName),
    make_set(AccountName)
    by
    InitiatingProcessParentFileName,
    InitiatingProcessFileName,
    InitiatingProcessCommandLine,
    ProcessCommandLine
```
